#+title: cadquery org-babel language

Reusable Python behind the =cadquery= org-babel language (see
=KONIX_ob-cadquery.el=).  This file is ingested into the Library of Babel, so any
note can pull the block below in with a =<<cadquery>>= noweb reference.

* the compose plugin

  The block plugs =composeunion= / =composecut= / =composeintersect= onto
  =cq.Workplane= (see [[https://cadquery.readthedocs.io/en/latest/extending.html][Extending CadQuery]]).  They behave exactly like
  =union= / =cut= / =intersect= but also record, on the result, what the step is
  built from — so the fluent chain stays plain cadquery, no wrapper.

  Recording accumulates only within one step's chain: =_compose_reset()= (injected
  by the executor right after every =<<...>>= noweb inclusion) drops the head, so
  the next =composeX= on the just-included shape starts a fresh recording with
  that shape as the grey base.  Hence each block renders *its* step: base grey,
  unioned parts orange, removed material (=before ∩ tool=) translucent red.

  =cadquery_export= is the single entry point the language calls; a shape without
  a recording (e.g. =result.val()=) is exported as STL instead.

  =openscad= runs an openscad source and sews the stl it answers into a cadquery
  solid, so a part a scad library already draws needs no redrawing here;
  =nix_build= builds a nix expression and hands back its store path, which is how
  a note pins such a library without installing anything.  =bosl2= puts the two
  together for [[https://github.com/BelfrySCAD/BOSL2][BOSL2]], whose threads,
  bottle necks and caps a note then calls by name.

    import cadquery as cq

    # Recording lives in a plain module-global list -- no state on Workplane
    # objects, so only the documented "add a method to cq.Workplane" plugin
    # mechanism is used.  It survives plain ops (mix composecut and a hidden
    # plain cut freely); _compose_reset() (injected after each <<...>> noweb
    # inclusion) clears it so each step records only its own ops.
    _LOG = []

    def _compose_reset():
        _LOG.clear()

    def _compose_op(name, role):
        orig = getattr(cq.Workplane, name)
        def method(self, other, *a, **k):
            result = orig(self, other, *a, **k)
            if not _LOG:
                _LOG.append(("base", self))
            _LOG.append((role, other, self))   # operand, and shape before this op
            return result
        return method

    cq.Workplane.composeunion     = _compose_op("union", "add")
    cq.Workplane.composecut       = _compose_op("cut", "cut")
    cq.Workplane.composeintersect = _compose_op("intersect", "intersect")

    _GREY = cq.Color(0.7, 0.7, 0.7, 1.0)
    # `add' shows the new material, opaque.  A subtractive op is split into two
    # disjoint, contrasted, very translucent parts: what it removes from the
    # part, and the tool's reach beyond the part.
    _ADD_COLORS = [cq.Color(0.75, 0.2, 0.9, 1.0),    # violet
                   cq.Color(0.95, 0.35, 0.7, 1.0),   # pink
                   cq.Color(0.5, 0.2, 0.8, 1.0)]     # purple
    _A = 0.18
    _GHOST = 0.06
    _HOLLOW = {                          # (removed-from-part, tool-reach-beyond)
        "cut":       (cq.Color(0.9, 0.15, 0.15, _GHOST),  # red, very ghostly -- material cut away
                      cq.Color(0.9, 0.25, 0.2, _A)),      # red -- tool beyond the part
        "intersect": (cq.Color(0.15, 0.4, 0.95, _A),      # blue -- base clipped away
                      cq.Color(0.2, 0.85, 0.4, _A)),      # green -- tool beyond the part
    }

    def openscad(source, libraries=None):
        """Render an openscad SOURCE and hand back what it made, as a cadquery
        solid.  LIBRARIES, when given, is a directory put on OPENSCADPATH so the
        source can include what lives there.  openscad answers in stl, a bag of
        triangles, so each one becomes a face and the faces are sewn back into
        the one solid cadquery will cut and join."""
        import os, subprocess, tempfile
        work = tempfile.mkdtemp()
        scad, stl = (os.path.join(work, n) for n in ("part.scad", "part.stl"))
        with open(scad, "w") as handle:
            handle.write(source)
        env = dict(os.environ)
        if libraries:
            env["OPENSCADPATH"] = libraries
        subprocess.run(["nix-shell", "-p", "openscad", "--run",
                        "openscad -o %s %s" % (stl, scad)],
                       env=env, stderr=subprocess.DEVNULL, check=True)
        return _sew(stl)

    def _sew(stl):
        from OCP.RWStl import RWStl
        from OCP.TopoDS import TopoDS
        from OCP.BRepBuilderAPI import (
            BRepBuilderAPI_MakePolygon, BRepBuilderAPI_MakeFace,
            BRepBuilderAPI_Sewing, BRepBuilderAPI_MakeSolid,
        )
        poly = RWStl.ReadFile_s(stl)
        sewing = BRepBuilderAPI_Sewing(1e-6)
        for i in range(1, poly.NbTriangles() + 1):
            a, b, c = poly.Triangle(i).Get()
            face = BRepBuilderAPI_MakePolygon(
                poly.Node(a), poly.Node(b), poly.Node(c), True)
            sewing.Add(BRepBuilderAPI_MakeFace(face.Wire()).Face())
        sewing.Perform()
        shell = TopoDS.Shell_s(sewing.SewedShape())
        return cq.Workplane(obj=cq.Solid(BRepBuilderAPI_MakeSolid(shell).Solid()))

      src = fetchzip {

    def bosl2(call, *files):
        """Render a BOSL2 CALL and hand back the solid it makes.  FILES names the
        BOSL2 sources the call needs -- "threading.scad", "bottlecaps.scad" and so
        on -- "std.scad" being included whatever happens.  The library itself is
        pinned to a revision and built by nix, so nothing is installed."""
        head = "".join("include <BOSL2/%s>\n" % name
                       for name in ("std.scad",) + files)
        return openscad(head + call, libraries=nix_build(_BOSL2_NIX))

    class raw(str):
        """A value written into the scad as it stands, for a constant the
        language knows by name -- =BOTTOM=, =UP= -- rather than a string."""

    def _scad_value(value):
        if isinstance(value, raw):
            return str(value)
        if isinstance(value, bool):
            return "true" if value else "false"
        if isinstance(value, str):
            return '"%s"' % value
        if isinstance(value, (list, tuple)):
            return "[%s]" % ", ".join(_scad_value(v) for v in value)
        return repr(value)

    _PARTCAD_PACKAGE = """
    partcad: ">=0.8.0"
    dependencies:
      pub:
        type: git
        url: https://github.com/partcad/partcad-index.git
        revision: main
    """

    def partcad(path, **params):
        """A part of the PartCAD public registry, as a cadquery solid.  PATH is
        what the registry calls it, "//pub/std/metric/bosl2/bottlecaps:generic_b
        ottle_neck" and the like, and PARAMS what it is to be built with.  The
        package that points at the registry is written to a temporary directory,
        so asking for a part leaves nothing behind."""
        import os, tempfile
        import partcad as pc
        work = tempfile.mkdtemp()
        with open(os.path.join(work, "partcad.yaml"), "w") as handle:
            handle.write(_PARTCAD_PACKAGE)
        context = pc.init(work)
        return cq.Workplane(
            obj=context.get_part_cadquery(path, params=params or None))

    _BOSL2_WIKI = "https://github.com/BelfrySCAD/BOSL2/wiki/%s"

    class _Scad(object):
        """Every module of the scad FILES, reachable by its own name: what you
        call on this is written out as a call and rendered.  `doc' holds the wiki
        page of each file, which is where the modules and their arguments are
        listed, and printing this says the same."""

        def __init__(self, files):
            self._files = files
            self.doc = [_BOSL2_WIKI % name for name in files]

        def __repr__(self):
            return "BOSL2 %s -- see %s" % (", ".join(self._files),
                                           ", ".join(self.doc))

        def __getattr__(self, name):
            def call(*args, **kwargs):
                written = [_scad_value(a) for a in args]
                written += ["%s=%s" % (k, _scad_value(v))
                            for k, v in kwargs.items()]
                return bosl2("%s(%s);" % (name, ", ".join(written)),
                             *self._files)
            return call

    def scad(*files):
        """BOSL2's FILES as an object whose every module is a method, so a part
        the library already draws is one call and needs no wrapper written for
        it: =scad("bottlecaps.scad").generic_bottle_neck(id=26, ...)=.  What it
        can be asked for is what its wiki page lists, and `doc' links there."""
        return _Scad(files)

    def nix_build(expression):
        """Build a nix EXPRESSION and hand back the store path it made, so a note
        can pin what it needs without installing anything or leaving a trace."""
        import os, subprocess, tempfile
        path = os.path.join(tempfile.mkdtemp(), "expression.nix")
        with open(path, "w") as handle:
            handle.write(expression)
        return subprocess.run(["nix-build", "--no-out-link", path],
                              capture_output=True, text=True,
                              check=True).stdout.strip()

    def cadquery_export(obj, out, tolerance=0.1):
        if len(_LOG) > 1:
            base = _LOG[0][1]
            view = cq.Assembly()
            kept = base.intersect(obj)
            if kept.solids().vals():
                view.add(kept, color=_GREY, name="kept")
            added_left = obj.cut(base)              # all new material, split per add
            counts = {"add": 0, "cut": 0, "intersect": 0}
            for role, operand, before in _LOG[1:]:
                i = counts[role]
                counts[role] += 1
                if role == "add":
                    piece = added_left.intersect(operand)   # only the genuinely new material
                    added_left = added_left.cut(operand)
                    if piece.solids().vals():
                        view.add(piece, color=_ADD_COLORS[i % len(_ADD_COLORS)],
                                 name="add_%d" % i)
                    continue
                removed = (before.intersect(operand) if role == "cut"
                           else before.cut(operand))        # material taken from the part
                reach = operand.cut(before)                 # tool beyond the part
                rem_col, reach_col = _HOLLOW[role]
                if removed.solids().vals():
                    view.add(removed, color=rem_col, name="%s_%d_removed" % (role, i))
                if reach.solids().vals():
                    view.add(reach, color=reach_col, name="%s_%d_reach" % (role, i))
            out = out.rsplit(".", 1)[0] + ".glb"
            view.export(out)
        else:
            cq.exporters.export(obj, out, angularTolerance=tolerance)
        return out
